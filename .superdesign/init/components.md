## Dynamic palette cards

const $ = id => document.getElementById(id);
const rgbOf = hex => [1,3,5].map(i=>parseInt(hex.slice(i,i+2),16));
function hslOf(hex){let [r,g,b]=rgbOf(hex).map(v=>v/255),mx=Math.max(r,g,b),mn=Math.min(r,g,b),d=mx-mn,l=(mx+mn)/2,h=0,s=0;if(d){s=d/(1-Math.abs(2*l-1));h=(mx===r?((g-b)/d+6)%6:mx===g?(b-r)/d+2:(r-g)/d+4)*60;}return[h,s,l]}
colors.forEach(c=>c.hsl=hslOf(c.hex));
const wheelColors=[...colors].sort((a,b)=>a.hsl[0]-b.hsl[0]);
$('shade-count').textContent=colors.length;
$('wheel').setAttribute('aria-label',`Spinning wheel of ${colors.length} named colors`);
$('footer-count').textContent=`COLOR STORY / ${colors.length} NAMED SHADES`;
const step=360/wheelColors.length;
$('wheel').style.background='conic-gradient('+wheelColors.map((c,i)=>`${c.hex} ${i*step}deg ${(i+1)*step}deg`).join(',')+')';
let current=colors.find(c=>c.name==='Okinawa Rust'),rotation=0,spinning=false;
const angleFor = c => -(wheelColors.indexOf(c)+.5)*step;
const reduced=()=>window.matchMedia('(prefers-reduced-motion: reduce)').matches;
function turnTo(c,spin){const goal=((angleFor(c)%360)+360)%360;rotation+=((goal-rotation%360)+360)%360+(spin?360*5:0);$('wheel').style.transform=`rotate(${rotation}deg)`;}
function notice(t){$('status').textContent=t;}
function nearest(h,s,l,excluded){return colors.filter(c=>!excluded.includes(c)).sort((a,b)=>score(a)-score(b))[0];function score(c){let dh=Math.abs(c.hsl[0]-h);dh=Math.min(dh,360-dh)/180;return dh*2+Math.abs(c.hsl[1]-s)*.45+Math.abs(c.hsl[2]-l)*.8;}}
function companions(c,mode){let [h,s,l]=c.hsl;let found=[c];let targets=mode==='neighbors'?[[h+30,s,l],[h-30,s,l]]:mode==='tonal'?[[h,s,Math.max(.15,l-.25)],[h,s,Math.min(.88,l+.25)]]:[[h+180,Math.max(.2,s),l],[h+180,s*.5,l>.5?.25:.72]];for(let [hh,ss,ll] of targets){let n=mode==='tonal'?colors.filter(x=>x.family===c.family&&!found.includes(x)).sort((a,b)=>Math.abs(a.hsl[2]-ll)-Math.abs(b.hsl[2]-ll))[0]:nearest((hh+360)%360,ss,ll,found);found.push(n)}return found;}
function renderPalette(){const list=companions(current,$('pairing').value);$('palette').replaceChildren(...list.map((c,i)=>{const b=document.createElement('button');b.className='palette-card';b.setAttribute('aria-label',`Explore ${c.name}, ${c.hex}`);b.disabled=spinning;b.innerHTML=`<span class="mini-swatch" style="background:${c.hex}"></span><span class="mini-info"><b>${c.name}</b><small>${i===0?'SELECTED':c.hex}</small></span>`;b.onclick=()=>{if(spinning)return;current=c;turnTo(c,false);render();notice(`Selected ${c.name}`)};return b}));}
function render(){const c=current;const [r,g,b]=rgbOf(c.hex);$('color-name').textContent=c.name;$('hex').textContent=c.hex;$('rgb').textContent=`RGB ${r}, ${g}, ${b}`;$('family-label').textContent=c.family;$('number').textContent=`${String(colors.indexOf(c)+1).padStart(3,'0')} / ${colors.length}`;$('color-block').style.background=c.hex;const lum=rgbOf(c.hex).map(v=>{v/=255;return v<=.04045?v/12.92:((v+.055)/1.055)**2.4}).reduce((a,v,i)=>a+v*[.2126,.7152,.0722][i],0);$('color-block').style.color=lum>.179?'#101114':'#ffffff';$('associations').textContent=c.associations.join(' · ');$('story-name').textContent=c.story;renderPalette();}
$('spin').onclick=()=>{if(spinning)return;const pool=colors.filter(c=>c!==current);let bytes=new Uint32Array(1);crypto.getRandomValues(bytes);const selected=pool[Math.floor(bytes[0]/4294967296*pool.length)];spinning=true;$('spin').disabled=true;$('spin').innerHTML='Finding a shade…';renderPalette();turnTo(selected,true);setTimeout(()=>{current=selected;spinning=false;$('spin').disabled=false;$('spin').innerHTML='Spin the wheel <span aria-hidden="true">↻</span>';render();notice(`${current.name}. ${current.associations.join(' and ')}.`)},reduced()?30:2850)};
